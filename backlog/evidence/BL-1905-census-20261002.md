## Census script (node, from the repo root)

A handler is a candidate when a constant derived from `__dirname` (directly
or through another such constant) is joined with one of the names above.
"spawns" only says the file starts some process; reading each file decides
whether it starts the REAL path.

```js
const fs=require('fs'),path=require('path');
const NAMES=['done_with_current.bb','done_with_current.sh','done_with_current_batch.bb','done_with_current_batch.sh','done_with_current_task.bb','done_with_current_task.sh','ready_for_next.bb','ready_for_next.sh','ready_for_next_batch.sh','ready_for_next_task.sh'];
const dir='specs/pipeline/steps';
for(const f of fs.readdirSync(dir).filter(f=>f.endsWith('.js')).sort()){
  const src=fs.readFileSync(path.join(dir,f),'utf8');
  // real-dir constants: any const whose initializer mentions __dirname, or derives from one that does
  const real=new Set();
  let grew=true;const consts=[...src.matchAll(/const\s+([A-Z_][A-Z0-9_]*)\s*=\s*([^;]+);/g)];
  while(grew){grew=false;for(const [,n,init] of consts){if(real.has(n))continue;if(/__dirname/.test(init)||[...real].some(r=>new RegExp('\\b'+r+'\\b').test(init))){real.add(n);grew=true;}}}
  const hits=[];
  for(const [,n,init] of consts){ if(!real.has(n))continue; for(const nm of NAMES){ if(init.includes(`'${nm}'`)) hits.push(`${n}=${nm}`);} }
  // inline joins on a real const
  for(const m of src.matchAll(/path\.join\(\s*([A-Z_][A-Z0-9_]*)\s*,([^)]*)\)/g)){ if(!real.has(m[1]))continue; for(const nm of NAMES) if(m[2].includes(`'${nm}'`)) hits.push(`inline ${m[1]}+${nm}`); }
  if(!hits.length)continue;
  const spawns=/(spawnSync|execFileSync|execSync|spawn|execFile)\s*\(/.test(src);
  const readsOnly=/readFileSync\(\s*path\.join\(\s*[A-Z_]+,\s*'(done_with|ready_for)/.test(src);
  console.log(f, '|', [...new Set(hits)].join(', '), '| spawns:',spawns);
}
```
