### Probe script: tmp/bl1711-external-probe.sh (run from the QA worktree root)

```bash
#!/usr/bin/env bash
# Scratch probe (stand-ins only, never port 11434): an EXTERNAL ollama server
# that is alive but not answering yet (e.g. loading a large model). Does the
# BL-1711 sweep start a second server? Run from the parcel's scratch tree root.
set -uo pipefail
LIB="${1:?path to ollama_ancillary_lib.sh}"
source "$LIB"
P="$(mktemp -d "${TMPDIR:-/tmp}/bl1711-ext-XXXXXX")"
mkdir -p "$P/.swarmforge" "$P/bin"
PORT=$(python3 -c 'import socket;s=socket.socket();s.bind(("127.0.0.1",0));print(s.getsockname()[1]);s.close()')
# the external server: a live process whose command line reads `ollama serve`
cat > "$P/bin/ollama" <<'SH'
#!/usr/bin/env bash
echo "started $* pid $$" >> "$(dirname "$0")/../starts.log"
sleep 300
SH
chmod +x "$P/bin/ollama"
"$P/bin/ollama" serve & ext=$!
sleep 0.5; : > "$P/starts.log"   # ignore the external's own start line
echo "external 'ollama serve' pid $ext alive: $(ps -o args= -p $ext)"
ollama_ancillary_write_record "$(ollama_ancillary_record_path "$P/.swarmforge")" external "" "" "http://127.0.0.1:$PORT"
export OLLAMA_ANCILLARY_CRASH_CONFIRM_SECONDS=1
for i in 1 2 3; do
  out=$(ollama_ancillary_restart_if_crashed "$P" "$P/bin/ollama" "$P/models" 4096 2 1 "$P/serve.log" 2>"$P/err.$i")
  echo "tick $i stdout: ${out:-<none>} | stderr: $(tr '\n' ' ' < "$P/err.$i")"
  sleep 1.2
done
echo "second-server starts while the external was alive: $(grep -c . "$P/starts.log")"
cat "$P/starts.log"
kill -0 $ext 2>/dev/null && echo "external pid $ext still alive (not signalled)"
pkill -P $$ sleep 2>/dev/null; kill $ext 2>/dev/null
pkill -f "$P/bin/ollama" 2>/dev/null
rm -rf "$P"
```
