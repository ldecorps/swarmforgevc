### repro_d1.js

```js
'use strict';
// QA repro for BL-1901 D1: build! ignores `git checkout --detach`'s exit.
const fs = require('node:fs');
const path = require('node:path');
const { spawnSync } = require('node:child_process');
const QA = '/home/carillon/swarmforgevc/.worktrees/QA';
const { fixture } = require(path.join(QA, 'specs/pipeline/steps/bl1872LanderDaemonSteps'));
const { makeFixture, git } = fixture;
const SCRIPT = path.join(QA, 'swarmforge/scripts/land_merge_path.bb');
const TICKET = 'BL-9001';
const fx = makeFixture();
try {
  const wt = path.join(fx.root, '.worktrees', 'lander');
  git(fx.root, 'worktree', 'add', '-q', '-B', 'swarmforge-lander', wt, 'origin/main');
  // the ticket's line, off origin/main
  git(fx.line, 'fetch', '-q', 'origin');
  git(fx.line, 'checkout', '-q', '--detach', 'origin/main');
  fs.writeFileSync(path.join(fx.line, 'bl9001.txt'), 'BL-9001 work\n');
  git(fx.line, 'add', 'bl9001.txt');
  git(fx.line, 'commit', '-q', '-m', `${TICKET}: own work`);
  const queued = git(fx.line, 'rev-parse', 'HEAD');
  // origin/main moves on after the line forked (another land), so build! must merge
  fs.writeFileSync(path.join(fx.root, 'other.txt'), 'other\n');
  git(fx.root, 'add', 'other.txt');
  git(fx.root, 'commit', '-q', '-m', 'BL-9002: another landed ticket');
  git(fx.root, 'push', '-q', 'origin', 'main');
  // an untracked, non-ignored file in the lander worktree at a path the queued commit tracks
  fs.writeFileSync(path.join(wt, 'bl9001.txt'), 'stray\n');
  const res = spawnSync('bb', [SCRIPT, wt, TICKET, queued], { encoding: 'utf8', timeout: 120000 });
  console.log('exit', res.status);
  console.log('--- stdout\n' + res.stdout + '--- stderr\n' + res.stderr);
  const main = git(fx.origin, 'rev-parse', 'main');
  console.log('origin main now', main, '|', git(fx.origin, 'log', '-1', '--format=%s', 'main'));
  console.log('queued', queued, 'ancestor of origin main?',
    spawnSync('git', ['-C', fx.origin, 'merge-base', '--is-ancestor', queued, 'main']).status === 0);
  console.log('origin main tree:', git(fx.origin, 'ls-tree', '--name-only', 'main').split('\n').join(' '));
} finally {
  fs.rmSync(fx.work, { recursive: true, force: true });
}
```
